function findKeyPath(obj, targetKey, path = []) {
  for (const key in obj) {
    const currentPath = path.concat(key);

    if (key === targetKey) {
      return currentPath;
    }

    if (typeof obj[key] === "object" && obj[key] !== null) {
      const result = findKeyPath(obj[key], targetKey, currentPath);
      if (result) {
        return result;
      }
    }
  }

  return null; // Key not found
}

const memoizedFunction=(fn)=>{
    const memory = new Map();

    memoize=(...args)=>{
        const input = JSON.stringify(args);
        if(memory.has(input)) return memory.get(input);
        const result= fn.call(this, args);
        memory.set(input, result);
        return result;
    }
    memoize.clear=()=>{
        memory.clear();
    }

    return memoize
}

// 12. debounce and  throttle
const debounce = function (fun, delay){
    let timer;
    return function(...arg){
        clearTimeout(timer)
        timer = setTimeout(()=>{
            fun.apply(arg)
        }, delay)
    }
}

const throttle = function(fun, delay){
    let flag= true;
    return function(...arg){
        if(flag){
            fun(...arg);
            flag=false;
             setTimeout(()=>{
                flag=true;
            }, delay)
        }    
    }
}

// 13. Convert a JavaScript object into a JSON string without using JSON.stringify().

const customStringify = (obj) => {
    if (obj === null) return "null";
    if (typeof obj === "string") return `"${obj}"`;
    if (typeof obj === "number" || typeof obj === "boolean") return obj.toString();
    if (Array.isArray(obj)) {
        return `[${obj.map(customStringify).join(",")}]`;
    }
    if (typeof obj === "object") {
        const keys = Object.keys(obj)
            .map(key => `"${key}":${customStringify(obj[key])}`)
            .join(",");
        return `{${keys}}`;
    }
    return undefined; // Handle unsupported types like functions
};

//14. Pormise.all

function promiseAll(promises) {
    return new Promise((resolve, reject) => {
      if (!Array.isArray(promises)) {
        return reject(new TypeError("Argument must be an array of promises"));
      }
  
      let results = [];
      let completedPromises = 0;
  
      if (promises.length === 0) {
        return resolve([]);
      }
  
      promises.forEach((promise, index) => {
        Promise.resolve(promise)
          .then((value) => {
            results[index] = value;
            completedPromises++;
  
            if (completedPromises === promises.length) {
              resolve(results);
            }
          })
          .catch(reject); // If any promise rejects, reject the whole Promise.all
      });
    });
  }
  
  const p1 = Promise.resolve(10);
const p2 = Promise.resolve(20);
const p3 = new Promise((resolve) => setTimeout(() => resolve(30), 1000));

promiseAll([p1, p2, p3])
  .then((results) => console.log("Results:", results)) // Output: Results: [10, 20, 30]
  .catch((error) => console.error("Error:", error));
